        /* LOAD USER */ 
 
        async function loadUser() { 
 
            const snapshot = 
                await get( 
                    ref(db, "users/" + currentUser.uid) 
                ); 
 
            if (snapshot.exists()) { 
 
                userData = 
                    snapshot.val(); 
 
            } else { 
 
                userData = { 
 
                    uid: currentUser.uid, 
 
                    name: currentUser.email.split("@")[0], 
 
                    email: currentUser.email, 
 
                    ffid: "", 
 
                    balance: 0, 
 
                    wins: 0, 
 
                    prizeWon: 0, 
 
                    joinedCount: 0 
 
                }; 
 
                await set( 
                    ref( 
                        db, 
                        "users/" + currentUser.uid 
                    ), 
                    userData 
                ); 
 
            } 
 
            renderProfile(); 
            updateDashboard(); 
 
        }

        window.joinTournament = async function() { 
 
            const id = 
                $("joinTournamentId").value; 
 
            const t = 
                tournaments[id]; 
 
            if (!t) return; 
 
 
            const fee = 
                Number(t.entryFee || 0); 
 
            const balance = 
                Number(userData.balance || 0); 
 
 
            if (balance < fee) { 
 
                toast("Insufficient balance."); 
 
                return; 
 
            } 
 
 
            /* CHECK DUPLICATE */ 
 
            const already = 
                Object.values(joins) 
                .some( 
                    x => 
                    x.uid === currentUser.uid && 
                    x.tournamentId === id && 
                    x.status !== "rejected" 
                ); 
 
 
            if (already) { 
 
                toast("You already joined."); 
 
                closeModal("joinModal"); 
 
                return; 
 
            } 
 
 
            /* DEDUCT ENTRY */ 
 
            const newBalance = 
                balance - fee; 
 
            await set( 
                ref( 
                    db, 
                    "users/" + currentUser.uid + "/balance" 
                ), 
                newBalance 
            ); 
 
 
            /* CREATE JOIN REQUEST */ 
 
            const joinRef = 
                push( 
                    ref(db, "joinRequests") 
                ); 
 
 
            await set( 
                joinRef, { 
 
                    uid: currentUser.uid, 
 
                    playerName: userData.name, 
 
                    email: currentUser.email, 
 
                    ffid: userData.ffid, 
 
                    tournamentId: id, 
 
                    tournamentName: t.name, 
 
                    entryFee: fee, 
 
                    status: "pending", 
 
                    createdAt: Date.now() 
 
                } 
            ); 
 
 
            /* UPDATE COUNT */ 
 
            await set( 
                ref( 
                    db, 
                    "users/" + currentUser.uid + "/joinedCount" 
                ), 
                Number(userData.joinedCount || 0) + 1 
            ); 
 
 
            userData.balance = 
                newBalance; 
 
            userData.joinedCount = 
                Number(userData.joinedCount || 0) + 1; 
 
            updateDashboard(); 
 
            toast("Tournament joined!"); 
 
            closeModal("joinModal"); 
 
        };

        window.submitMoney = async function() { 
 
            const type = 
                $("moneyType").value; 
 
            const amount = 
                Number($("moneyAmount").value); 
 
            const reference = 
                $("moneyReference").value.trim(); 
 
 
            if (!amount || amount <= 0) { 
 
                toast("Enter valid amount."); 
 
                return; 
 
            } 
 
 
            if (!reference) { 
 
                toast("Reference required."); 
 
                return; 
 
            } 
 
 
            /* WITHDRAWAL BALANCE CHECK */ 
 
            if ( 
                type === "withdrawal" && 
                amount > Number(userData.balance || 0) 
            ) { 
 
                toast("Insufficient balance."); 
 
                return; 
 
            } 
 
 
            const request = 
                push( 
                    ref(db, "moneyRequests") 
                ); 
 
 
            await set( 
                request, { 
 
                    uid: currentUser.uid, 
 
                    email: currentUser.email, 
 
                    name: userData.name, 
 
                    type: type, 
 
                    amount: amount, 
 
                    reference: reference, 
 
                    status: "pending", 
 
                    createdAt: Date.now() 
 
                } 
            ); 
 
 
            toast( 
                type === "deposit" ? 
                "Deposit request submitted." : 
                "Withdrawal request submitted." 
            ); 
 
            closeModal("moneyModal"); 
 
        };
